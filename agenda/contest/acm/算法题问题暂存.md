[P4408 [NOI2003\] 逃学的小孩 - 洛谷](https://www.luogu.com.cn/problem/P4408)

为什么以下代码无法求出离直径端点最远的点

```cpp
ci M = (int)2e5+3;
int n,m,h[M],v[2*M],N[2*M],e[2*M],tot,f[M],p1,p2,lu[M];
lld zj,sum[M],d[M],ans;
bool vis[M];
void add(int x,int y,int z)
{
    v[++tot]=y;e[tot]=z;N[tot]=h[x];h[x]=tot;
}
void dfs1(int x,int fa)
{
    if(zj<d[x])
    {
        p1=x;zj=d[x];
    }
    for(int i=h[x];i;i=N[i])
    {
        int y=v[i];int z=e[i];
        if(y==fa||vis[y])ct;
        d[y]=d[x]+z;
        dfs1(y,x);
    }
}
void dfs2(int x,int fa)
{
    if(zj<d[x])
    {
        p2=x;zj=d[x];
    }
    for(int i=h[x];i;i=N[i])
    {
        int y=v[i];int z=e[i];
        if(y==fa)ct;
        d[y]=d[x]+z;
        f[y]=i;
        dfs2(y,x);
    }
}
void solve()
{
    cin>>n>>m;tot=1;
    repp(i,1,m)
    {
        lld u,v,t;cin>>u>>v>>t;
        add(u,v,t);add(v,u,t);
    }
    dfs1(1,0);
    d[p1]=0;zj=0;
    dfs2(p1,0);
    int l=p2;int r=p1;int len=0;
    int fen;ans=zj;lld tt;int chuzj=zj;
    tot=0;
    while(l!=r)
    {
        vis[l]=1;lu[++tot]=l;sum[tot+1]=sum[tot]+e[f[l]];
        l=v[f[l]^1];
    }
    vis[r]=1;
    repp(i,1,tot)
    {
        tt=min(sum[i],sum[tot]-sum[i]);
        d[lu[i]]=0;zj=0;
        dfs1(lu[i],0);
        tt+=zj;
        ans=max(ans,chuzj+tt);
    }
    cout<<ans;
    ret;
}
signed main()
{
    ios::sync_with_stdio(0);cin.tie(0);
    solve();
    ret 0;
}
```
